cmake_minimum_required(VERSION 3.20)
project(bold_capture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The supplied Insta360 SDK requires Windows x64")
endif()
foreach(SDK CAMERA_SDK MEDIA_SDK)
  if(NOT IS_DIRECTORY "${${SDK}}/include")
    message(FATAL_ERROR "Set ${SDK} to the extracted SDK directory")
  endif()
endforeach()
add_executable(bold_capture capture.cpp)
target_compile_definitions(bold_capture PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(bold_capture PRIVATE "${CAMERA_SDK}/include" "${MEDIA_SDK}/include/stitcher")
target_link_libraries(bold_capture PRIVATE "${CAMERA_SDK}/lib/CameraSDK.lib" "${MEDIA_SDK}/lib/MediaSDK.lib" ws2_32)
target_compile_options(bold_capture PRIVATE /utf-8 /EHsc)
